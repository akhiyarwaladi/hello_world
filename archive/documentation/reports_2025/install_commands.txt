# 1. Aktivasi base environment
conda activate base

# 2. Navigasi ke folder project
cd "C:\Users\MyPC PRO\Documents\hello_world"

# 3. Install PyTorch dengan CUDA 12.8 (ini yang terpenting!)
pip install torch==2.8.0 torchvision==0.23.0 --index-url https://download.pytorch.org/whl/cu128

# 4. Verifikasi CUDA
python -c "import torch; print(f'PyTorch: {torch.__version__}'); print(f'CUDA Available: {torch.cuda.is_available()}'); print(f'CUDA Version: {torch.version.cuda if torch.cuda.is_available() else \"N/A\"}')"

# 5. Install Ultralytics (YOLO)
pip install ultralytics==8.3.202

# 6. Install dependencies dari requirements.txt
pip install -r requirements.txt

# 7. Verifikasi semua package
python -c "import torch; print(f'✓ PyTorch: {torch.__version__}')"
python -c "import ultralytics; print(f'✓ Ultralytics: {ultralytics.__version__}')"
python -c "import cv2; print(f'✓ OpenCV: {cv2.__version__}')"
python -c "import pandas; print(f'✓ Pandas: {pandas.__version__}')"
